         CMDS -> CMD MAIS_CMDS

     CMD_COND -> IF ESQPAREN CONDICAO DIRPAREN ESQCHAVE CMDS DIRCHAVE PFALSA
               | WHILE ESQPAREN CONDICAO DIRPAREN ESQCHAVE CMDS DIRCHAVE
          CMD -> COUT INSERCAO ID INSERCAO ENDL
               | ID RESTO_IDENT
       PFALSA -> ELSE ESQCHAVE CMDS DIRCHAVE
               | ε
  RESTO_IDENT -> ATRIBUICAO EXP_IDENT
    EXP_IDENT -> EXPRESSAO
               | LER_DOUBLE ESQPAREN DIRPAREN
     CONDICAO -> EXPRESSAO RELACAO EXPRESSAO

               | DES
               | MAI
               | MI
               | MA
               | ME
    EXPRESSAO -> TERMO OUTROS_TERMOS
        TERMO -> OP_UN FATOR MAIS_FATORES
        OP_UN -> MENOS
               | ε
        FATOR -> ID
               | NUMERO_REAL
               | ESQPAREN EXPRESSAO DIRPAREN

               | ε
        OP_AD -> MAIS
               | MENOS
 MAIS_FATORES -> OP_MUL FATOR MAIS_FATORES
               | ε
       OP_MUL -> MULT
               | DIV
